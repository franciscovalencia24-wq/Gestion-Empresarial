        function handleImageUpload(event) {

            }
        }

            } else {
                // MODO NORMAL
                secNormal.classList.remove('hidden');
                secNormal.classList.add('block');
                secPremium.classList.add('hidden');
                secPremium.classList.remove('block');
                boxSimple.classList.remove('hidden'); // Mostrar "En Simple"

                document.getElementById('ui-simple-text').innerText = currentData.enSimple;
                document.getElementById('ui-importance').innerText = currentData.importancia;
                document.getElementById('ui-footer-note').innerText = currentData.notaFooter || '';
                
                document.getElementById('ui-importance').className = `text-[10px] font-black italic uppercase ${config.impactText}`;
                document.getElementById('ui-simple-box').className = `mt-4 p-4 rounded-lg border ${config.box} ${config.border}`;
                document.getElementById('ui-simple-header').className = `flex items-center gap-2 mb-1 font-bold text-xs uppercase ${config.text}`;

                const tableBody = document.getElementById('ui-table-body');
                tableBody.innerHTML = (currentData.impactos || []).map(i => `<tr><td>${i.plazo}</td><td>${i.texto}</td></tr>`).join('');
            }

            // Renderizar Iconos Lucide (necesario tras inyectar HTML)
            lucide.createIcons();
        }

        function updateData() {
            try {
                const input = document.getElementById('data-input').value.trim();
                if (!input) return;
                const newData = JSON.parse(input);
                currentData = { ...currentData, ...newData };
                render();
            } catch (e) {
                alert("Error en el formato JSON. Revisa que el código esté bien copiado.");
            }
        }

        async function exportImage() {
            const area = document.getElementById('capture-area');
            try {
                const canvas = await html2canvas(area, { 
                    scale: 3, 
                    useCORS: true,
                    backgroundColor: currentMode === 'premium' ? '#000000' : '#ffffff',
                    logging: false
                });
                const link = document.createElement('a');
                link.download = `FV_Infografia_${currentMode.toUpperCase()}_${Date.now()}.png`;
                link.href = canvas.toDataURL('image/png');
                link.click();
            } catch (err) {
                alert("Error al exportar la imagen.");
            }
        }

        window.onload = () => {
            document.getElementById('data-input').value = JSON.stringify(currentData, null, 2);
            render();
        };
